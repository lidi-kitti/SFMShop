-- Анализ существующей структуры
-- Таблица orders нормализована:
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id),
    total DECIMAL(10, 2)
);


-- Отчёт «заказы за день с именами покупателей»
SELECT o.id, u.name, o.total FROM orders o JOIN users u ON u.id = o.user_id
WHERE o.created_at >= CURRENT_DATE;

--можно сделать денормализацию, чтобы каждый раз не делать запрос с JOIN
-- Триггер 1: при создании/смене владельца заказа подтягиваем имя из users.
-- Без него orders.user_name у новых заказов останется NULL.
CREATE OR REPLACE FUNCTION fill_order_user_name()
RETURNS TRIGGER AS $$
BEGIN
    SELECT name INTO NEW.user_name FROM users WHERE id = NEW.user_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER fill_orders_user_name
BEFORE INSERT OR UPDATE OF user_id ON orders
FOR EACH ROW
EXECUTE FUNCTION fill_order_user_name();

-- Триггер 2: при переименовании пользователя распространяем имя на все его заказы.
-- Обновление денормализованного поля при изменении заказа
CREATE OR REPLACE FUNCTION update_orders_user_name()
RETURNS TRIGGER AS $$
BEGIN
 UPDATE orders SET user_name = NEW.name WHERE user_id = NEW.id;
 RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_orders_on_user_change
AFTER UPDATE ON orders
FOR EACH ROW
WHEN (OLD.name IS DISTINCT FROM NEW.name)
EXECUTE FUNCTION update_orders_user_name();

-- Аналогично для reviews
CREATE TABLE reviews (
    id SERIAL PRIMARY KEY,
    product_id INTEGER REFERENCES products(id),
    product_name VARCHAR(200),  -- Денормализация: для ускорения отчетов
    user_id INTEGER REFERENCES users(id),
    review_text TEXT,
    rating INTEGER
);

-- Отчёт «отзывы с названиями товаров»
SELECT r.rating, p.name, r.review_text FROM reviews r JOIN products p ON p.id = r.product_id
WHERE r.created_at >= CURRENT_DATE - INTERVAL '7 days';
--можно сделать денормализацию, чтобы каждый раз не делать запрос с JOIN

-- Триггер 1: при создании/смене товара в отзыве подтягиваем имя из products.
CREATE OR REPLACE FUNCTION fill_review_product_name()
RETURNS TRIGGER AS $$
BEGIN
    SELECT name INTO NEW.product_name FROM products WHERE id = NEW.product_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER fill_reviews_product_name
BEFORE INSERT OR UPDATE OF product_id ON reviews
FOR EACH ROW
EXECUTE FUNCTION fill_review_product_name();

-- Триггер 2: при переименовании товара распространяем имя на все отзывы.
-- Обновление денормализованного поля при изменении товара
CREATE OR REPLACE FUNCTION update_review_product_name()
RETURNS TRIGGER AS $$
BEGIN
 UPDATE reviews SET product_name = NEW.name WHERE product_id = NEW.id;
 RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_reviews_on_product_change
AFTER UPDATE ON products
FOR EACH ROW
WHEN (OLD.name IS DISTINCT FROM NEW.name)
EXECUTE FUNCTION update_review_product_name();

--компромиссы между нормализацией и денормализацией
нормализация:
Нет дублирования данных. Легче обновлять данные. Меньше места для хранения. Лучшая целостность данных.

денормализация:
Быстрее запросы (меньше JOIN). Проще запросы. Лучше для отчетов.
Сложная бизнес-логика — её сложно отлаживать, тяжело тестировать, легко устроить циклические триггеры.
«Невидимые» побочные эффекты — другой разработчик не поймёт, почему UPDATE products каким-то чудом меняет reviews. Минимум — добавляй комментарий к триггеру: COMMENT ON TRIGGER ... IS '...'.
